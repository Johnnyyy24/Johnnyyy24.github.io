{{- /* Markdown image hook: shrink large images, convert to WebP, add width/height and lazy loading. GIFs (may be animated) and non-processable formats are served as-is. */ -}} {{- $u := urls.Parse .Destination -}} {{- $res := "" -}} {{- if not $u.IsAbs -}} {{- $res = .Page.Resources.Get $u.Path -}} {{- end -}} {{- $src := .Destination -}} {{- $w := 0 -}}{{- $h := 0 -}} {{- with $res -}} {{- if eq .ResourceType "image" -}} {{- if and (ne .MediaType.SubType "gif") (ne .MediaType.SubType "svg") -}} {{- $img := . -}} {{- if gt .Width 1600 -}} {{- $img = .Resize "1600x webp q80" -}} {{- else -}} {{- $img = .Process "webp q80" -}} {{- end -}} {{- $src = $img.RelPermalink -}}{{- $w = $img.Width -}}{{- $h = $img.Height -}} {{- else if eq .MediaType.SubType "gif" -}} {{- $src = .RelPermalink -}}{{- $w = .Width -}}{{- $h = .Height -}} {{- else -}} {{- $src = .RelPermalink -}} {{- end -}} {{- else -}} {{- $src = .RelPermalink -}} {{- end -}} {{- end -}} {{- /**/ -}}
